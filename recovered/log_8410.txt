class AutoLabelWorker(QObject):
    progress = Signal(int, int)
    finished = Signal()
    failed = Signal(str)

    def __init__(self, records, settings) -> None:
        super().__init__()
        self.records = records
        self.settings = settings
        self.cancelled = False

    def run(self) -> None:
        try:
            if not self.settings.onnx_model_path:
                raise ValueError("璇峰厛鍦ㄩ」鐩缃腑閫夋嫨 ONNX 妯″瀷")
            detector = YoloOnnxDetector()
            detector.load(self.settings.onnx_model_path)
            total = len(self.records)
            for index, record in enumerate(self.records, start=1):
                if self.cancelled:
                    break
                from PIL import Image
                with Image.open(record.path) as image:
                    predictions = detector.predict(
                        image,
                        self.settings.label_presets,
                        self.settings.input_size,
                        self.settings.confidence_threshold,
                        self.settings.nms_threshold,
                    )
                record.annotations = predictions
                record.status = "labeled" if predictions else "unlabeled"
                self.progress.emit(index, total)
            self.finished.emit()
        except Exception as exc:
            self.failed.emit(str(exc))
