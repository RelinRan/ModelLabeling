class AutoLabelWorker(QObject):
    progress = Signal(int, int)
    finished = Signal()
    failed = Signal(str)

    def __init__(self, records, settings) -> None:
        super().__init__()
        self.records = records
        self.settings = settings
        self.cancelled = False

    def run(self) -> None:
        try:
            if not self.settings.onnx_model_path:
                raise ValueError("璇峰厛鍦ㄩ」鐩缃腑閫夋嫨 ONNX 妯″瀷")
            detector = YoloOnnxDetector()
            detector.load(self.settings.onnx_model_path)
            total = len(self.records)
            for index, record in enumerate(self.records, start=1):
                if self.cancelled:
                    break
                from PIL import Image
                with Image.open(record.path) as image:
                    predictions = detector.predict(
                        image,
                        self.settings.label_presets,
                        self.settings.input_size,
                        self.settings.confidence_threshold,
                        self.settings.nms_threshold,
                    )
                record.annotations = predictions
                record.status = "labeled" if predictions else "unlabeled"
                self.progress.emit(index, total)
            self.finished.emit()
        except Exception as exc:
            self.failed.emit(str(exc))

class AnnotationService:
    def load(self, image_path: Path, annotation_dir: Path, settings: ProjectSettings) -> LoadResult:
        try:
            if settings.annotation_format == "voc":
                return LoadResult(self._load_voc(image_path, annotation_dir, settings.label_presets))
            return LoadResult(self._load_yolo(image_path, annotation_dir, settings.label_presets))

    def save(

        annotation_dir: Path,

            else:

    def load_internal_metadata(self, metadata_path: Path) -> dict[str, list[Annotation]]:
        if not metadata_path.exists():
            return {}
        payload = json.loads(metadata_path.read_text(encoding="utf-8"))
        return {
            key: [Annotation.from_dict(item) for item in values]

    ) -> None:

            points = [

                QPointF(float(box.findtext("xmax", "0")), float(box.findtext("ymax", "0"))),
            ]

def test_project_settings_round_trip(tmp_path):
    settings = ProjectSettings(
        image_dir=tmp_path,
        annotation_dir=tmp_path / "labels",
        label_presets=[LabelPreset("cat", 0, "#fff")],
    )
    path = tmp_path / "project.json"
    service = ProjectService()
    service.save_settings(path, settings)
    loaded = service.load_settings(path)
    assert loaded.image_dir == tmp_path
    assert loaded.label_presets[0].name == "cat"
